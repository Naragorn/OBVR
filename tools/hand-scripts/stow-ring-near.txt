# The inventory ring shown faintly while a held item is near it (vr/PlaceNear.h, camera/hands/Held.cpp
# StepStowRingNear; [Hands] StowSpotVisible, [Hands] PlaceHintOpacity; Nadi, 2026-11-16: "das selbe auch
# für den inv stow spot ... sehr dezent"). potion-belt.txt's Restore Health potion pulled from the floor
# (a weapon would be taken at its grip and equipped, not held): held down there and held up and away the
# ring is not shown; brought to the chest it comes ("the ring shown"), taken away again it goes ("the
# ring hidden"), back at the chest it comes again, and let go there the potion goes into the pack and
# the ring goes with it.
# Run with: python3 tools/hand-script-run.py --headless tools/hand-scripts/stow-ring-near.txt

ini Hands Enabled=1
ini Hands StowAtBody=1
ini Hands StowSpotVisible=1
ini Hands PlaceHintOpacity=0.25
# The ring where the defaults have it, whatever the player's INI says (the first run, 2026-11-16: the
# tester's own ring sat 0.16 m further back and the hand "at the chest" was 0.29 m from its middle).
ini Hands StowRight=0
ini Hands StowForward=0.17
ini Hands StowUp=-0.33
ini Hands StowRadius=0.16
ini Debug EyeDump=1
console-at settle CloseAllMenus
console player.additem 00009311 1
console player.drop 00009311 1

expect HandScript: finished
count 2 Stow ring near: the ring shown (an item held
count 2 Stow ring near: the ring hidden (an item held
expect of it -, opacity 0.25)
expect Stow: the held item is at the body
expect Stow: let go at the body - taken (no owner
reject HandScript: WARNING
reject Stow: let go at the body - not taken
reject Stow: let go at the body - refused

wait 15
mark settle
wait 1
head pitch -50
right pos 0.00 -1.22 -0.73
right rot -80 0 0
wait 1.5
mark near
right grip 0.9
wait 4
mark held
# Held up and well away from the chest: no ring.
head pitch 0
right pos 0.20 0.10 -0.50
wait 1
mark held_away
# To the chest: the ring comes. The controller 0.22 m under where stow.txt had it: the ring is measured
# at the drawn hand, which this controller's turn puts that much above it (run 20261116-095711: "0.06
# right, 0.03 forward, 0.22 up of it").
right pos -0.03 -0.52 -0.15
wait 1
mark at_ring
# Away: it goes.
right pos 0.20 0.10 -0.50
wait 1
mark away_again
# Back, and let go there: stowed, the ring gone with the item.
right pos -0.03 -0.52 -0.15
wait 1
mark at_ring_again
right grip 0
wait 1.5
mark stowed
wait 0.5
