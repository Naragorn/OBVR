# A message box over the open tablet: the inventory is up on the tablet, then
# a message box is raised by the console's MessageBox (merchant-barter.txt does
# the same over a barter). The tablet carries a message (vr::TabletMenusUp), the
# world pauses for it as vanilla (vr::TabletMenuRunsWorld). Wanted: the page
# read with 1 popup, "Menu pause" lines for the Message menu.
# Run with: python3 tools/hand-script-run.py tools/hand-scripts/tablet-message.txt --headless --no-steamvr --parallel

ini Hands Enabled=1
ini Tablet Enabled=1
ini Tablet Unroll=vertical
ini Hands LeftHandGripX=0
ini Hands LeftHandGripY=0
ini Hands LeftHandGripZ=0
ini Hands HandGripUpMetres=0
ini Hands HandGripForwardMetres=0
ini Tablet InScene=1
ini Render UnpausedMenus=0
ini Render MonitorMenusOnly=0
ini Debug EyeDump=1
console-at inventory MessageBox "OBVR tablet test" "Yes" "No"
console-at box ClickMenuButton "#4" 1001

expect HandScript: finished
expect Tablet: the Inventory page read
expect Message box: "OBVR tablet test"
expect Menu pause
reject HandScript: WARNING

wait 12
left pos -0.12 -0.32 -0.32
left rot 0 0 0
right pos 0.15 -0.25 -0.15
right rot 40 25 0
wait 1.5
left b 1
wait 0.2
left b 0
wait 2
mark inventory
wait 3
mark box
wait 2
mark after
