# Stowing with the hand held out in front and moving as it opens (the first
# headset test, 2026-09-27: a sword let go 0.23 m ahead of the eyes stayed on
# the floor). Otherwise as stow.txt: an Iron Longsword dropped
# on the floor ahead, the right grip closed near it (it floats to the hand),
# the hand brought to the chest, the grip let go there - the sword goes into
# the inventory the way activating it would take it. The inventory opened
# afterwards shows it.
# Run with: tools\hand-script-run.ps1 -Script tools\hand-scripts\stow-in-front.txt

ini Hands Enabled=1
ini Hands StowAtBody=1
ini Render MirrorMenusToMonitor=1
console player.additem 00000C0C 1
console player.drop 00000C0C 1

expect HandScript: finished
expect Stow: the held item is at the body
expect Stow: let go at the body - taken (no owner
reject HandScript: WARNING
reject Stow: let go at the body - not taken
reject Stow: let go at the body - refused

wait 10
head pitch -50
right pos 0.05 -1.40 -0.80
right rot -80 0 0
wait 1
mark near
dump
right grip 1
wait 1.5
mark held
dump
# To the chest, the head level again.
head pitch 0
right pos 0.05 -0.40 -0.23
wait 1
mark in_front
# Moving away as the grip opens: 0.1 s out of the zone, then let go.
right pos 0.05 -0.40 -0.45
wait 0.1
right grip 0
wait 1
mark stowed
right pos 0.15 -0.25 -0.35
right rot 62 29 0
left b 1
wait 0.2
left b 0
wait 1.5
mark inventory
left b 1
wait 0.2
left b 0
wait 1
