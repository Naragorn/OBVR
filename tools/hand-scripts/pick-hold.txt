# The pick held steady (game/PickHold.h): two Iron Longswords dropped
# together on the floor ahead, the right hand brought down over them and
# swept slowly across from one to the other and back. The pick should take
# one, stay on it while the hand wavers between them, and change only when
# the other is clearly the nearer - a few "Pick: on" lines over the sweep,
# not one every frame. The ring, the info text and the outline follow the
# held one. Read the count of "Pick: on" lines in the log after the run.
# Run with: tools\hand-script-run.ps1 -Script tools\hand-scripts\pick-hold.txt

ini Hands Enabled=1
ini Hands TargetOutline=1
# No hand bodies: a hand pushing the swords apart would move the thing measured.
ini Hands BodyCollision=0

console player.additem 00000C0C 2
console player.drop 00000C0C 1
console player.drop 00000C0C 1

expect HandScript: finished
expect Pick: on
reject HandScript: WARNING

# The swords settle first; the left hand hangs down, its laser at the floor behind.
