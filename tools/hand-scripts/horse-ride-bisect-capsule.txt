ini Debug HorseTurnProbe=1
ini Hands BodyRadiusScale=1
ini Hands ReachOpens=0
console-at ready player.moveto 0002008C
console-at ready SetNumericINISetting "fAutoVanityModeDelay:Camera" 100000

console-at priory player.setangle z 0
console-at priory tai
console-at aimed tai
console-at mounted tai

expect HandScript: finished
expect Camera: the camera node's ancestry
expect the player has mounted
expect Camera: riding at the mark
expect a world frame while the game loads

wait 12
