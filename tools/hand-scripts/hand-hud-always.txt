# The hand HUD is shown always by default since 2026-10-14 ([HandHud]
# ShowOnLook=0, vr::HandHudLookedAt, docs/hud-on-hands-spec.md): the same
# moves as hand-hud-look.txt - each palm turned to the face, then away - and
# no look counts, so no "looked at" line and nothing fades. The "Hand HUD: at"
# lines show the hand HUD was up at those marks. ShowOnLook is left out on
# purpose: the run checks the default, so a live INI with ShowOnLook=1 fails
# it. The quads on the hands need the controllers' device indices, which a
# run with the controllers off does not have; the headset shows those.
# Run with: python3 tools/hand-script-run.py tools/hand-scripts/hand-hud-always.txt
#      or:  tools\hand-script-run.ps1 -Script tools\hand-scripts\hand-hud-always.txt

ini Hands Enabled=1
ini HandHud Enabled=1

expect HandScript: finished
expect Hand HUD: at "left_yaw_plus" - Bars
expect Hand HUD: at "right_yaw_minus" - Bars
reject Hand HUD: the left hand is looked at
reject Hand HUD: the right hand is looked at
reject HandScript: WARNING

wait 10
right pos 0.30 -0.45 -0.20
left pos -0.30 -0.45 -0.20
wait 1
mark rest
left pos 0.00 -0.15 -0.35
left rot 0 90 0
wait 1
mark left_yaw_plus
left rot 0 -90 0
wait 1
mark left_yaw_minus
left pos -0.30 -0.45 -0.20
left rot 0 0 0
right pos 0.00 -0.15 -0.35
right rot 0 90 0
wait 1
mark right_yaw_plus
right rot 0 -90 0
wait 1
mark right_yaw_minus
wait 1
