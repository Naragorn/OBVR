# Teleport with TeleportMakesNoise=1: while the glide moves, the player's movement
# flags read "walking forward" to the detection only (what NPCs hear); the
# player must not walk on by them after the landing.

ini Hands Enabled=1
ini Locomotion Teleport=1
ini Locomotion TeleportInstant=0
ini Locomotion TeleportMakesNoise=1
ini Locomotion TeleportSound=swish

expect HandScript: finished
expect Teleport: arrived
reject HandScript: WARNING
expect Teleport: a second after the landing the player stands
expect Teleport: the landing sound swish (00088834) played
reject Teleport: the landing sound swish (00088834) NOT played
reject Sound: 

wait 4
mark standing
# The right hand forward, pointing a little up: the arc lands ahead.
right pos 0.2 -0.3 -0.35
right rot 15 0 0
wait 0.5
right stick 0 1
wait 0.8
mark aiming
dump
wait 0.2
right stick 0 0
wait 2
mark arrived
dump
wait 1
