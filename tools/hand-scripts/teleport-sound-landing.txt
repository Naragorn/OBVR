# Teleport with TeleportMakesNoise=1 and TeleportSound=landing: the landing
# plays the game's own landing sound for the ground under the player
# (0x006B1900 with the controller's material) and nothing breaks.

ini Hands Enabled=1
ini Locomotion Teleport=1
ini Locomotion TeleportInstant=0
ini Locomotion TeleportMakesNoise=1
ini Locomotion TeleportSound=landing

expect HandScript: finished
expect Teleport: arrived
reject HandScript: WARNING
expect Teleport: a second after the landing the player stands
expect Teleport: the landing sound landing (00000000) played
reject Teleport: the landing sound landing (00000000) NOT played
reject Sound: 

wait 4
mark standing
# The right hand forward, pointing a little up: the arc lands ahead.
right pos 0.2 -0.3 -0.35
right rot 15 0 0
wait 0.5
right stick 0 1
wait 0.8
mark aiming
dump
wait 0.2
right stick 0 0
wait 2
mark arrived
dump
wait 1
