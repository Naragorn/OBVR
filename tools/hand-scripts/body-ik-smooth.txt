# The body IK's smoothness (the tester, 2026-10-12: "ruckelt und zuckelt der
# körper und alles auf einmal. keine flüssige erfahrung!"): Debug.BodyTrace
# logs the body every frame - while the inputs stand still (the hands
# hanging at about the arm's length, held out at the reach's edge, the right
# one before the chest at the torso's keep-out), while that hand slides an
# inch at a time, while the player walks (the walk animation under the
# pinned body), while the head pitches in steps and while it crouches.
# tools\body-trace.ps1 reads the frame-to-frame differences out of the
# run's log: with the inputs still, nothing of the body may move from one
# frame to the next. The eye dumps show the body looked down at.
# Run with: tools\hand-script-run.ps1 -Script tools\hand-scripts\body-ik-smooth.txt

ini Hands Enabled=1
ini Body Visible=1
ini Body IK=1
ini Body IKArms=1
ini Body IKTorso=1
ini Body IKLegs=1
ini Debug BodyTrace=4000
